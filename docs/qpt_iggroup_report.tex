\section{Quantum Process Tomography: Overview}

\subsection{What is QPT?}

\textbf{Quantum Process Tomography (QPT)} is a characterization method that
completely describes how a quantum system transforms under a physical process
(gate, noise channel, circuit). Unlike state tomography which characterizes
a single quantum state, QPT characterizes the \emph{map} itself.

Formally, any linear, completely positive, trace-preserving (CPTP) map
$\mathcal{E}: \mathcal{H}^n \to \mathcal{H}^n$ admits the
\textbf{Kraus representation}~\cite{kraus1983states}:
\begin{equation}
  \mathcal{E}(\rho) = \sum_k K_k \rho K_k^\dagger,
  \qquad \sum_k K_k^\dagger K_k = I.
\end{equation}

\subsection{The Choi Matrix}

A compact representation of $\mathcal{E}$ is the
\textbf{Choi matrix}~\cite{choi1975completely}:
\begin{equation}
  \Lambda_\mathcal{E} = (I_n \otimes \mathcal{E})(|\Phi^+\rangle\langle\Phi^+|),
\end{equation}
where $|\Phi^+\rangle = \frac{1}{\sqrt{2^n}}\sum_{i}|i\rangle|i\rangle$
is the maximally entangled state on $2n$ qubits. For a unitary $U$:
\begin{equation}
  \Lambda_U = U \otimes U^*.
\end{equation}
This is a $2n$-qubit matrix, requiring $16^n \times 16$\,bytes in dense storage.
For $n=8$: $\approx 281$\,TB -- prohibitive for any dense simulator.

\subsection{Applications of QPT}

\begin{itemize}
  \item \textbf{Noise characterization}: determine Kraus operators of a noisy
    gate from experiment, enabling error mitigation.
  \item \textbf{Gate certification}: compute gate fidelity
    $F = \frac{1}{4^n}\mathrm{Tr}(\Lambda_{\mathrm{ideal}}^\dagger \Lambda_{\mathrm{exp}})$.
  \item \textbf{Reduced-state analysis}: compute subsystem state via partial trace
    $\rho_B = \mathrm{Tr}_A(\Lambda)$.
  \item \textbf{Quantum error correction design}: characterize residual errors
    after syndrome measurements.
\end{itemize}

\section{Decision Diagram Representation of Choi Matrices}

\subsection{QMDD Background}

A \textbf{Quantum Multi-valued Decision Diagram (QMDD)}~\cite{miller2006qmdd}
represents a $2^n \times 2^n$ complex matrix as a directed acyclic graph.
For structured matrices, QMDD node count grows polynomially in $n$.

\subsection{Choi DD Construction}

\begin{algorithm}[h]
\caption{Build Choi Matrix DD}
\begin{algorithmic}[1]
\Procedure{BuildChoiDD}{circuit $\mathcal{C}$, package $\mathcal{P}$}
  \State $U \gets \mathcal{P}.\texttt{buildFunctionality}(\mathcal{C})$
  \State $U^\dagger \gets \mathcal{P}.\texttt{conjugateTranspose}(U)$
  \State $U^* \gets \mathcal{P}.\texttt{transpose}(U^\dagger)$
  \State $\Lambda \gets \mathcal{P}.\texttt{kronecker}(U,\; U^*)$
  \State \Return $\Lambda$ \Comment{$2n$-qubit Choi matrix DD}
\EndProcedure
\end{algorithmic}
\end{algorithm}

\section{igGroupSifting for Choi Matrix Compression}

\subsection{Mirror Symmetry of Choi Matrices}

$\Lambda_U = U \otimes U^*$ has a natural symmetry: qubit $i$ and qubit $i+n$
are mirrors of each other. Any two-qubit gate $(a,b)$ in $U$ creates
simultaneous interactions $(a,b)$ in the upper half and $(a{+}n, b{+}n)$ in
the lower half. Standard Sifting~\cite{rudell1993sifting} ignores this and
may destroy the structure, inflating the DD.

\subsection{igGroupSifting Algorithm}

\textbf{igGroupSifting}~\cite{iggroup2024} detects symmetric variable groups
via an Interaction Graph and moves them together during reordering:

\begin{enumerate}
  \item Build IG: $w(i,j) = $ \#two-qubit gates between qubits $i,j$.
  \item Detect symmetric pairs: $i \sim j$ iff $\forall k: w(i,k)=w(j,k)$.
  \item Sift symmetric groups as units, preserving structural symmetry.
\end{enumerate}

For the Choi system we construct a \emph{Choi-specific IG}:

\begin{algorithm}[h]
\caption{Build Choi IG}
\begin{algorithmic}[1]
\Procedure{BuildChoiIG}{circuit $\mathcal{C}$, $n$}
  \State $G.\texttt{init}(2n)$
  \For{each gate $g$ involving qubits $\{a,b,\ldots\} \subseteq [0,n)$}
    \State add edges $(a,b)$ and $(a{+}n, b{+}n)$
  \EndFor
  \For{$q = 0$ \textbf{to} $n-1$}
    \State add mirror coupling $(q,\; q{+}n)$ \Comment{encode symmetry}
  \EndFor
  \State $G.\texttt{detectSymmetry}()$
  \State \Return $G$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\subsection{End-to-End QPT Pipeline}

\begin{algorithm}[h]
\caption{QPT Pipeline with igGroupSifting}
\begin{algorithmic}[1]
\Procedure{QPT-IGGroupSifting}{$\mathcal{C}$, $n$}
  \State $\Lambda \gets \texttt{BuildChoiDD}(\mathcal{C})$
    \Comment{Build $2n$-qubit Choi DD}
  \State $G \gets \texttt{BuildChoiIG}(\mathcal{C}, n)$
    \Comment{Build Choi-specific IG}
  \State $\Lambda' \gets \texttt{igGroupSifting}(\Lambda, G)$
    \Comment{Compress}
  \State $\rho_B \gets \mathrm{Tr}_{\mathrm{upper}}(\Lambda')$
    \Comment{Partial trace: reduced density matrix}
  \State \Return $\rho_B$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\section{Summary of Contributions}

\begin{enumerate}
  \item \textbf{Choi DD construction}: $\Lambda_U = U \otimes U^*$ via
    \texttt{kronecker}, extending the QMDD package to QPT.
  \item \textbf{Choi-specific IG}: mirror coupling construction encodes
    the $q_i \leftrightarrow q_{i+n}$ symmetry, enabling igGroupSifting
    to detect and exploit it.
  \item \textbf{End-to-end QPT benchmark}: peak nodes and total time across
    gate types (H\_layer, CNOT\_chain, QFT, Grover, Clifford), $n=2$--$8$.
  \item \textbf{Stability result}: igGroupSifting avoids Sifting's destructive
    reordering (Grover $n=8$: Sifting $+105\%$ nodes; igGroupSifting $0\%$)
    while matching its compression quality on symmetric circuits.
\end{enumerate}
